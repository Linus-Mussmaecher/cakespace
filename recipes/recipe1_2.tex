\section*{Lemonade Cake}

\begin{tabular}{rl}
	\toprule
	Amount & Ingredient \\
	\midrule
	\qty{630}{g}    & Flour \\
	\qty{425}{g}    & Sugar \\
	\qty{30}{g}      & Vanilla Sugar \\
	\qty{22}{g}    & Baking Powder \\
	\qty{0.25}{tsp} & Salt \\
	\qty{185}{g}    & Oil \\
	\qty{550}{g}    & Orange Lemonade \\
	\midrule
	\qty{360}{g}    & Powdered Sugar \\
	\qty{60}{g}     & Citrus Juice \\
	\bottomrule
\end{tabular}

\subsection*{Instructions}
\begin{enumerate}
	\item Pre-heat oven to \qty{190}{\degreeCelsius}.
	\item Mix flour, sugar, vanilla sugar, baking powder and salt.
	\item Slows add oil and orange lemonade while stirring.
	\item Put on tray and bake for \qty{25}{min}.
	\item Mix powdered sugar and citrus juice to cover top of baked cake.
\end{enumerate}
